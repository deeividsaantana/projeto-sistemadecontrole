Private Sub TratarAtalhoTeclado(ByRef KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If (Shift And 2) <> 0 And KeyCode = vbKeyS Then
        KeyCode = 0
        SalvarPorAtalho
        Exit Sub
    End If

    If KeyCode = vbKeyEscape Then
        KeyCode = 0
        cmdFechar_Click
        Exit Sub
    End If
End Sub

Private Sub CarregarPrefixos()
    Dim ws As Worksheet
    Dim r As Long
    Dim ultima As Long

    Set ws = ThisWorkbook.Worksheets("Equipamentos")
    ultima = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    cmbPrefixo.Clear
    For r = 2 To ultima
        If Trim(ws.Cells(r, "A").Value) <> "" Then cmbPrefixo.AddItem ws.Cells(r, "A").Value
    Next r
End Sub

    cmbCombustivel.Value = COMBUSTIVEL_PADRAO

Private Sub DefinirMesPadrao()
    Select Case Month(Date)
        Case 6: cmbMes.Value = "JUNHO 2026"
        Case 7: cmbMes.Value = "JULHO 2026"
        Case 8: cmbMes.Value = "AGOSTO 2026"
        Case 9: cmbMes.Value = "SETEMBRO 2026"
        Case 10: cmbMes.Value = "OUTUBRO 2026"
        Case Else: If cmbMes.ListCount > 0 Then cmbMes.ListIndex = cmbMes.ListCount - 1
    End Select
End Sub

Private Sub BuscarEquipamento()
    Dim ws As Worksheet
    Dim achado As Range
    Dim prefixo As String

    prefixo = Trim(cmbPrefixo.Value)
    txtDescricao.Value = ""
    txtEmpresa.Value = ""
    If prefixo = "" Then Exit Sub

    Set ws = ThisWorkbook.Worksheets("Equipamentos")
    Set achado = ws.Columns("A").Find(What:=prefixo, LookIn:=xlValues, LookAt:=xlWhole)

    If achado Is Nothing Then
        txtDescricao.Value = "Não cadastrado"
        txtEmpresa.Value = "Não cadastrado"
        lblStatus.Caption = "Prefixo não encontrado."
    Else
        txtDescricao.Value = ws.Cells(achado.Row, "C").Value
        txtEmpresa.Value = ws.Cells(achado.Row, "I").Value
        lblStatus.Caption = "Equipamento localizado."
    End If
End Sub

Private Sub AtualizarBombas()
    Dim ws As Worksheet
    Dim bombaInicial As Variant
    Dim litros As Double

    txtBombaInicial.Value = ""
    txtBombaFinal.Value = ""

    If cmbMes.Value = "" Then Exit Sub
    If cmbComboio.Value = "" Then Exit Sub

    Set ws = ThisWorkbook.Worksheets(cmbMes.Value)
    bombaInicial = UltimaBombaFinal(ws, cmbComboio.Value)

    If bombaInicial <> "" Then txtBombaInicial.Value = bombaInicial

    If EhNumeroFormulario(txtLitros.Value) And txtBombaInicial.Value <> "" Then
        litros = NumeroFormulario(txtLitros.Value)
        txtBombaFinal.Value = NumeroFormulario(txtBombaInicial.Value) + litros
    End If
End Sub

Private Function UltimaBombaFinal(ws As Worksheet, comboio As String) As Variant
    Dim r As Long
    Dim ultima As Long

    ultima = ws.Cells(ws.Rows.Count, "I").End(xlUp).Row

    For r = ultima To 6 Step -1
        If Trim(ws.Cells(r, "I").Value) = Trim(comboio) Then
            If ws.Cells(r, "M").Value <> "" Then
                UltimaBombaFinal = ws.Cells(r, "M").Value
                Exit Function
            End If
        End If
    Next r

    UltimaBombaFinal = ""
End Function

Private Sub cmdSalvar_Click()
    Dim ws As Worksheet
    Dim linha As Long

    If Not NormalizarCampoHora(True) Then Exit Sub
    If Not ValidarFormulario Then Exit Sub

    Set ws = ThisWorkbook.Worksheets(cmbMes.Value)
    linha = ProximaLinhaLivre(ws)

    ws.Cells(linha, "B").Value = CDate(txtData.Value)
    ws.Cells(linha, "C").Value = cmbPrefixo.Value
    ws.Cells(linha, "D").Value = txtDescricao.Value
    ws.Cells(linha, "E").Value = NumeroFormulario(txtKM.Value, 0)
    ws.Cells(linha, "F").Value = NumeroFormulario(txtHorimetro.Value, 0)
    ws.Cells(linha, "G").Value = NumeroFormulario(txtLitros.Value)
    ws.Cells(linha, "H").Value = TimeValue(txtHora.Value)
    ws.Cells(linha, "H").NumberFormat = "hh:mm"
    ws.Cells(linha, "I").Value = cmbComboio.Value
    ws.Cells(linha, "J").Value = COMBUSTIVEL_PADRAO
    ws.Cells(linha, "K").Value = txtEmpresa.Value
    ws.Cells(linha, "L").Value = NumeroFormulario(txtBombaInicial.Value)
    ws.Cells(linha, "M").Value = NumeroFormulario(txtBombaFinal.Value)

    Application.CalculateFull
    MsgBox "Lançamento salvo na linha " & linha & ".", vbInformation
    LimparCampos
    FocarPrefixo
End Sub

Private Function ValidarFormulario() As Boolean
    ValidarFormulario = False

    If cmbMes.Value = "" Then MsgBox "Selecione o mês.", vbExclamation: Exit Function
    If Not IsDate(txtData.Value) Then MsgBox "Informe uma data válida.", vbExclamation: Exit Function
    If cmbPrefixo.Value = "" Then MsgBox "Selecione o prefixo.", vbExclamation: Exit Function
    If cmbComboio.Value = "" Then MsgBox "Selecione o comboio.", vbExclamation: Exit Function

    If Not IsNumeric(txtLitros.Value) Or NumeroFormulario(txtLitros.Value) <= 0 Then
        MsgBox "Informe litros maior que zero.", vbExclamation
        Exit Function
    End If

    If txtBombaInicial.Value = "" Then MsgBox "Bomba inicial não encontrada para este comboio.", vbExclamation: Exit Function
    If txtBombaFinal.Value = "" Then MsgBox "Bomba final não calculada.", vbExclamation: Exit Function

    ValidarFormulario = True
End Function

Private Function NumeroFormulario(ByVal valor As Variant, Optional ByVal padrao As Double = 0) As Double
    Dim texto As String

    texto = Trim(CStr(valor))
    texto = Replace(texto, "'", "")
    texto = Replace(texto, "´", "")
    texto = Replace(texto, "`", "")
    texto = Replace(texto, " ", "")

    If Len(texto) = 0 Then
        NumeroFormulario = padrao
        Exit Function
    End If

    If Application.DecimalSeparator = "," Then
        texto = Replace(texto, ".", ",")
    Else
        texto = Replace(texto, ",", ".")
    End If

    If Not IsNumeric(texto) Then Err.Raise 13
    NumeroFormulario = CDbl(texto)
End Function

Private Function EhNumeroFormulario(ByVal valor As Variant) As Boolean
    On Error GoTo Invalido
    NumeroFormulario valor
    EhNumeroFormulario = True
    Exit Function
Invalido:
    EhNumeroFormulario = False
End Function
Private Function ProximaLinhaLivre(ws As Worksheet) As Long
    Dim r As Long

    For r = 6 To ws.Rows.Count
        If Trim(ws.Cells(r, "B").Value) = "" And _
           Trim(ws.Cells(r, "C").Value) = "" And _
           Trim(ws.Cells(r, "G").Value) = "" Then
            ProximaLinhaLivre = r
            Exit Function
        End If
    Next r
End Function

Private Function Nz(valor As Variant, Optional padrao As Variant = 0) As Variant
    If Trim(CStr(valor)) = "" Then Nz = padrao Else Nz = valor
End Function

Private Function FormatarHoraDigitada(ByVal valor As String) As String
    Dim texto As String
    Dim partes() As String
    Dim hora As Long
    Dim minuto As Long

    texto = Trim(CStr(valor))

    If Len(texto) = 0 Then
        FormatarHoraDigitada = ""
        Exit Function
    End If

    texto = Replace(texto, ".", ":")
    texto = Replace(texto, " ", "")

    If InStr(1, texto, ":", vbTextCompare) > 0 Then
        partes = Split(texto, ":")
        If UBound(partes) <> 1 Then Err.Raise 5
        If Not IsNumeric(partes(0)) Or Not IsNumeric(partes(1)) Then Err.Raise 5
        hora = CLng(partes(0))
        minuto = CLng(partes(1))
    Else
        texto = Replace(texto, "-", "")
        texto = Replace(texto, "/", "")
        If Not IsNumeric(texto) Then Err.Raise 5

        If Len(texto) <= 2 Then
            hora = CLng(texto)
            minuto = 0
        Else
            texto = Right("0000" & texto, 4)
            hora = CLng(Left(texto, Len(texto) - 2))
            minuto = CLng(Right(texto, 2))
        End If
    End If

    If hora < 0 Or hora > 23 Or minuto < 0 Or minuto > 59 Then Err.Raise 5

    FormatarHoraDigitada = Format(TimeSerial(hora, minuto, 0), "hh:nn")
End Function

Private Function NormalizarCampoHora(Optional ByVal mostrarAviso As Boolean = True) As Boolean
    On Error GoTo HoraInvalida

    txtHora.Value = FormatarHoraDigitada(txtHora.Value)
    NormalizarCampoHora = True
    Exit Function

HoraInvalida:
    NormalizarCampoHora = False
    If mostrarAviso Then MsgBox "Digite a hora no formato 1005 ou 10:05.", vbExclamation
    On Error Resume Next
    txtHora.SetFocus
    On Error GoTo 0
End Function

End Sub
