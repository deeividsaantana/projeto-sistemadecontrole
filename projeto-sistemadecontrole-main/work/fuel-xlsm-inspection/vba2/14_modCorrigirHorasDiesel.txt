Option Explicit

Public Sub CorrigirHorasColunaH_Diesel()
    Dim nomes As Variant, nome As Variant
    Dim ws As Worksheet
    Dim ultima As Long, r As Long
    Dim v As Variant, novo As Variant

    On Error GoTo Finalizar
    Application.ScreenUpdating = False
    Application.EnableEvents = False

    nomes = Array("JUNHO 2026", "JULHO 2026", "AGOSTO 2026", "SETEMBRO 2026", "OUTUBRO 2026")

    For Each nome In nomes
        Set ws = Nothing
        On Error Resume Next
        Set ws = ThisWorkbook.Worksheets(CStr(nome))
        On Error GoTo Finalizar

        If Not ws Is Nothing Then
            ultima = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row
            If ultima < 6 Then ultima = ws.Cells(ws.Rows.Count, "C").End(xlUp).Row
            If ultima > 5000 Then ultima = 5000

            If ultima >= 6 Then
                For r = 6 To ultima
                    v = ws.Cells(r, "H").Value2
                    If Trim(CStr(v)) <> "" Then
                        novo = HoraCorrigidaDiesel(v)
                        If Not IsEmpty(novo) Then
                            ws.Cells(r, "H").Value = CDbl(novo)
                            ws.Cells(r, "H").NumberFormat = "hh:mm"
                        End If
                    End If
                Next r
            End If
            ws.Range("H6:H" & Application.Max(ultima, 6)).NumberFormat = "hh:mm"
        End If
    Next nome

Finalizar:
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    ThisWorkbook.Save
End Sub

Private Function HoraCorrigidaDiesel(ByVal valor As Variant) As Variant
    Dim texto As String, partes() As String
    Dim n As Double, inteiro As Long, h As Long, m As Long

    If IsEmpty(valor) Then Exit Function
    texto = Trim(CStr(valor))
    texto = Replace(texto, "'", "")
    texto = Replace(texto, "´", "")
    texto = Replace(texto, "`", "")
    texto = Replace(texto, " ", "")
    If texto = "" Then Exit Function

    If InStr(1, texto, ":", vbTextCompare) > 0 Then
        partes = Split(texto, ":")
        If UBound(partes) < 1 Then Exit Function
        If Not IsNumeric(partes(0)) Or Not IsNumeric(partes(1)) Then Exit Function
        h = CLng(partes(0)): m = CLng(partes(1))
    ElseIf IsNumeric(valor) Then
        n = CDbl(valor)
        If n > 0 And n < 1 Then HoraCorrigidaDiesel = n: Exit Function
        If n > 2359 Or n <> Fix(n) Then Exit Function
        inteiro = CLng(n)
        h = inteiro \ 100: m = inteiro Mod 100
        If inteiro < 100 Then h = inteiro: m = 0
    Else
        texto = Replace(texto, ",", ".")
        If Not IsNumeric(texto) Then Exit Function
        n = CDbl(texto)
        If n > 0 And n < 1 Then HoraCorrigidaDiesel = n: Exit Function
        If n > 2359 Or n <> Fix(n) Then Exit Function
        inteiro = CLng(n)
        h = inteiro \ 100: m = inteiro Mod 100
        If inteiro < 100 Then h = inteiro: m = 0
    End If

    If h < 0 Or h > 23 Or m < 0 Or m > 59 Then Exit Function
    HoraCorrigidaDiesel = CDbl(TimeSerial(h, m, 0))
End Function
